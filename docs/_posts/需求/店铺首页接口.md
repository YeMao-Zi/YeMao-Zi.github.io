4.店铺首页-店铺活动入口区

get /mobile/shop/shop-store-decoration/getStoreActivityArea

入参：

```
export interface Request {
    /**
     * 店铺ID
     */
    store_id?: number;
    [property: string]: any;
}
```

返回：

```
export interface Response {
    /**
     * 店铺活动区数据
     */
    areas: Area[];
    /**
     * 是否有装修配置
     */
    hasDecoration: boolean;
    /**
     * 店铺ID
     */
    storeId: number;
    [property: string]: any;
}

export interface Area {
    /**
     * 活动入口(areaType=4)
     */
    activityEntry: ActivityEntry;
    /**
     * 区域名称
     */
    areaName: string;
    /**
     * 区域类型: 1-企业档案 2-优惠券 3-图片宣传 4-活动入口 5-热销 6-新品 7-推荐
     */
    areaType: number;
    [property: string]: any;
}

/**
 * 活动入口(areaType=4)
 */
export interface ActivityEntry {
    /**
     * 多件折活动，数据结构类型与秒杀活动一致，这里不再重复
     */
    multiDiscount: { [key: string]: any };
    /**
     * 秒杀活动
     */
    seckill: Seckill;
    [property: string]: any;
}

/**
 * 秒杀活动
 */
export interface Seckill {
    /**
     * 活动结束时间
     */
    activityEndTime: string;
    /**
     * 活动ID
     */
    activityId: number;
    /**
     * 活动名称
     */
    activityName: string;
    /**
     * 活动开始时间
     */
    activityStartTime: string;
    /**
     * 活动状态
     */
    activityState: number;
    /**
     * 活动标题
     */
    activityTitle: string;
    /**
     * 该活动下商品的最低折扣(活动价/原价)
     */
    minDiscount: number;
    /**
     * 参与活动的商品列表
     */
    products: Product[];
    /**
     * 多件折阶梯规则列表
     */
    reductionRules: SeckillReductionRule[];
    [property: string]: any;
}

export interface Product {
    /**
     * 商品活动价格(BigDecimal)
     */
    activityItemPrice: number;
    /**
     * 商品限时折扣
     */
    discount: number;
    /**
     * 商品SKU ID(用于跳转商品详情)
     */
    itemId: number;
    /**
     * 商品销售价格(SKU 维度 item_unit_price)
     */
    itemUnitPrice: number;
    /**
     * 起订量
     */
    minOrderQuantity: number;
    /**
     * 商品ID
     */
    productId: number;
    /**
     * 商品主图
     */
    productImage: string;
    /**
     * 商品名称
     */
    productName: string;
    /**
     * 该商品参与的多件折阶梯规则,仅多件折活动(1133)返回
     */
    reductionRules: ProductReductionRule[];
    /**
     * 商品 SKU 累计销量
     */
    sales: number;
    [property: string]: any;
}

export interface ProductReductionRule {
    /**
     * 具体折扣数值(= max_num / 10,保留 1 位小数),前端拼接"折"展示为"9折"等
     */
    discount: number;
    /**
     * 折扣比例,例如 90=9折、80=8折
     */
    maxNum: number;
    /**
     * 满足规则需购买的数量
     */
    total: number;
    [property: string]: any;
}

export interface SeckillReductionRule {
    /**
     * 具体折扣数值(= max_num / 10,保留 1 位小数),前端拼接"折"展示为"9折"等
     */
    discount: number;
    /**
     * 折扣比例,例如 90=9折、80=8折
     */
    maxNum: number;
    /**
     * 满足规则需购买的数量
     */
    total: number;
    [property: string]: any;
}
```
