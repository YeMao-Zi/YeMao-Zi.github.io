// ═══════════════ Language ═══════════════
var currentLang = 'zh';

function switchLang(lang) {
  currentLang = lang;
  document.querySelectorAll('.lang-btn').forEach(function(btn) {
    btn.classList.toggle('active', btn.getAttribute('data-lang') === lang);
  });
  document.querySelectorAll('[data-' + lang + ']').forEach(function(el) {
    var text = el.getAttribute('data-' + lang);
    if (text) el.textContent = text;
  });
  document.querySelectorAll('[data-placeholder-' + lang + ']').forEach(function(el) {
    var ph = el.getAttribute('data-placeholder-' + lang);
    if (ph) el.placeholder = ph;
  });
  document.documentElement.lang = lang === 'zh' ? 'zh-CN' : 'en';
}

// ═══════════════ Navigation ═══════════════
function scrollToPage(id) {
  var el = document.getElementById(id);
  if (el) el.scrollIntoView({ behavior: 'smooth', block: 'start' });
}

// ═══════════════ Form Helpers ═══════════════
function clearInput(id) {
  var input = document.getElementById(id);
  if (!input) return;
  input.value = '';
  input.classList.remove('has-value');
  input.focus();
}

function togglePwd(id, btn) {
  var input = document.getElementById(id);
  if (input.type === 'password') {
    input.type = 'text';
    btn.innerHTML = '&#128064;';
  } else {
    input.type = 'password';
    btn.innerHTML = '&#128065;';
  }
}

// ═══════════════ OTP Countdown ═══════════════
var otpSeconds = 63;
var otpInterval = setInterval(function() {
  otpSeconds--;
  var timerEl = document.getElementById('otpTimer');
  if (timerEl) timerEl.textContent = otpSeconds;
  if (otpSeconds <= 0) {
    clearInterval(otpInterval);
    if (timerEl) {
      timerEl.textContent = '0';
      timerEl.style.color = '#bfbfbf';
    }
  }
}, 1000);

// ═══════════════ Pill Toggle (for doc-pills) ═══════════════
document.addEventListener('click', function(e) {
  var pill = e.target.closest('.doc-pill');
  if (!pill) return;
  var input = pill.querySelector('input');
  if (!input) return;
  if (input.type === 'radio') {
    var name = input.name;
    pill.parentElement.querySelectorAll('.doc-pill').forEach(function(p) {
      p.classList.remove('checked');
    });
    pill.classList.add('checked');
    input.checked = true;
  } else if (input.type === 'checkbox') {
    pill.classList.toggle('checked');
    input.checked = !input.checked;
  }
});
